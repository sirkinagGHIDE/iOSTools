// TABATA TIMER v8.1 FIXED // No curly quotes version const fm = FileManager.local() const presetFile = fm.joinPath(fm.documentsDirectory(), 'tabata_presets.json') function loadPresets() { if (!fm.fileExists(presetFile)) { return { Default: { tabatas: 2, rounds: 8, work: 20, roundRest: 10, tabataRest: 60 } } } return JSON.parse(fm.readString(presetFile)) } const presets = loadPresets() let settings = { ...presets.Default } let paused = false let reset = false let isRunning = false const html = `
Tabata 0/0 - Round 0/0
READY
0
Log initialized...
` const web = new WebView() web.shouldAllowRequest = (request) => { return true } let messageQueue = [] web.onMessage = (msg) => { let action = typeof msg === 'object' ? (msg.body || msg) : msg console.log('Scriptable received: ' + action) messageQueue.push(action) } const sleep = (ms) => new Promise(res => Timer.schedule(ms, false, res)) async function countdown(sec, label, color, tabataNum, roundNum) { const status = 'Tabata ' + tabataNum + '/' + settings.tabatas + ' - Round ' + roundNum + '/' + settings.rounds for (let i = sec; i > 0; i--) { if (reset) return false await web.evaluateJavaScript('update("' + status + '", "' + label + '", "' + i + '", "' + color + '")') for (let j = 0; j < 10; j++) { await sleep(100) if (reset) return false while (paused) { await web.evaluateJavaScript('update("' + status + '", "PAUSED", "' + i + '", "#2c2c2e")') await sleep(200) if (reset) return false } } } return true } async function runWorkout() { if (isRunning) { console.log('Already running') return } isRunning = true reset = false paused = false console.log('Starting Workout...') try { for (let t = 1; t <= settings.tabatas; t++) { for (let r = 1; r <= settings.rounds; r++) { if (!await countdown(settings.work, 'WORK', '#1e4620', t, r)) throw 'RESET' if (r < settings.rounds) { if (!await countdown(settings.roundRest, 'REST', '#1a1a2e', t, r)) throw 'RESET' } } if (t < settings.tabatas) { if (!await countdown(settings.tabataRest, 'TABATA REST', '#2c1a1a', t, settings.rounds)) throw 'RESET' } } await web.evaluateJavaScript('update("Complete!", "DONE", "0", "#004d00")') console.log('Workout Complete!') } catch (e) { if (e === 'RESET') { await web.evaluateJavaScript('update("", "READY", "0", "black")') console.log('Reset') } } finally { isRunning = false reset = false paused = false } } async function processMessages() { while (true) { await sleep(100) while (messageQueue.length > 0) { const action = messageQueue.shift() console.log('Processing: ' + action) if (action === 'start') { if (!isRunning) runWorkout() } else if (action === 'pause') paused = true else if (action === 'resume') paused = false else if (action === 'reset') reset = true } } } async function main() { const a = new Alert() a.title = 'Select Preset' Object.keys(presets).forEach(p => a.addAction(p)) const idx = await a.present() if (idx === -1) return settings = presets[Object.keys(presets)[idx]] console.log('Settings: ' + JSON.stringify(settings)) Device.setIdleTimerDisabled(true) await web.loadHTML(html) await web.present(true) await sleep(500) await web.evaluateJavaScript('update("", "READY", "0", "black")') console.log('View Presented. Starting message processor...') await processMessages() } await main()
